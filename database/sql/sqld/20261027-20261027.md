# 서브쿼리
- SELECT STU_NAME AS 학생, GRADE AS 학년 FROM STUDENT WHERE STU_ID IN ( SELECT STU_ID FROM ENROLL WHERE SUB_ID = 'SUB03' );

- 동아리에 가입되지 않은 학생 (상관 쿼리, LEFT JOIN)
SELECT STU_NAME AS 학생, GRADE AS 학년 FROM STUDENT S WHERE NOT EXISTS ( SELECT 1 FROM CLUB_MEMBER CM WHERE CM.STU_ID = S.STU_ID );

SELECT STU_NAME AS 학생, GRADE AS 학년 FROM STUDENT S LEFT JOIN CLUB_MEMBER CM ON CM.STU_ID = S.STU_ID WHERE CM.STU_ID IS NULL;

- 스칼라 서브쿼리
SELECT 
    SU.SUB_NAME AS 과목,
    ( SELECT COUNT(*) FROM ENROLL E WHERE E.SUB_ID = SU.SUB_ID ) AS 신청자수
FROM SUBJECT SU
ORDER BY 신청자수 DESC;

- 인라인 뷰 (FROM 절 안에 서브쿼리)
SELECT *
FROM ( 
    SELECT DEPT_ID, COUNT(*) AS 학생수
    FROM STUDENT
    WHERE DEPT_ID IS NOT NULL
    GROUP BY DEPT_ID
)
WHERE 학생수 >= 4
ORDER BY 학생수 DESC;

- SQL간 집합 연산자 ( UNION - 합집합 / UNION ALL / INTERSECT - 교집합 / MINUS - 차집합 )
SELECT SUB_ID FROM SUBJECT -- 등록 과목 (7개)
UNION
SELECT SUB_ID FROM ENROLL; -- 수강 과목 (30개)

**7개 - 중복 제거 (등록 과목은 수강 과목에 무조건 포함되니까)**

SELECT SUB_ID FROM SUBJECT
INTERSECT
SELECT SUB_ID FROM ENROLL; 

**과목 중 학생들이 실제로 수강하는 과목**

SELECT SUB_ID FROM SUBJECT
MINUS
SELECT SUB_ID FROM ENROLL;

# 소계 함수 
SELECT GRADE, GENDER, COUNT(*) FROM STUDENT
GROUP BY GRADE, GENDER
ORDER BY GRADE ASC, GENDER DESC;
-> (ROLLUP(), CUBE(), GROUPING SETS(), GROUPING)

SELECT GRADE, GENDER, COUNT(*) FROM STUDENT
GROUP BY ROLLUP(GRADE, GENDER)
ORDER BY GRADE ASC, GENDER DESC;

**합계를 낼 때는 ROLLUP - 지정해준 함수의 소개와 전체 합계**

SELECT GRADE, GENDER, COUNT(*) FROM STUDENT
GROUP BY CUBE(GRADE, GENDER)
ORDER BY GRADE ASC, GENDER DESC;

**CUBE는 앞의거의 집계, 뒤에거의 집계 등 모든 나올 수 있는 합계 다 내줌**
**= ROLLUP (GRADE, GENDER), (GRADE), (GENDER), ()**

SELECT GRADE, GENDER, COUNT(*) FROM STUDENT
GROUP BY GROUPING SET ( (GRADE), (GENDER) )
ORDER BY GRADE ASC, GENDER DESC;

# 윈도우 함수
**(윈도우 함수) OVER (PARTIION BY, ORDER BY)**
- 구분을 나누어서 RANK, SUM, 어쩌구 저쩌구우 하는 거!
- 파트 안에서 처리하는 방법

SELECT ENR_ID, MIDTERM,
    RANK()       OVER (ORDER BY MIDTERM DESC) AS RANK순위, -- 동점자 있으면 숫자 스킵 있음
    DENSE_RANK() OVER (ORDER BY MIDTERM DESC) AS DENSE순위, -- 동점자 있으면 숫자 스킵 없음
    ROW_NUMBER() OVER (ORDER BY MIDTERM DESC) AS LINE행번호
FROM SCORE WHERE MIDTERM IS NOT NULL
ORDER BY MIDTERM DESC;
**동점자 처리가 달라짐**

SELECT DEPT_ID AS 학과, TEACHER_NAME AS 교사, SALARY AS 급여,
    RANK() OVER (PARTITION BY DEPT_ID ORDER BY SALARY DESC) AS 학과내순위
FROM TEACHER
ORDER BY DEPT_ID, 학과내순위;

SELECT DEPT_ID AS 학과, TEACHER_NAME AS 교사, SALARY AS 급여,
    ROUND(AVG(SALARY) OVER (PARTITION BY DEPT_ID), 1) AS 학과평균,
    COUNT(*) OVER (PARTITION BY DEPT_ID) AS 학과인원
FROM TEACHER
ORDER BY DEPT_ID, SALARY DESC;

SELECT ENR_ID AS 신청번호, MIDTERM AS 중간,
    SUM(MIDTERM) OVER (ORDER BY ENR_ID) AS 누적합
FROM SCORE WHERE MIDTERM IS NOT NULL
ORDER BY ENR_ID;

- 나를 기준으로 앞뒤 데이터 가져오기
SELECT TEACHER_NAME AS 교사, SALARY AS 급여,
    LAG(SALARY) OVER (ORDER BY SALARY DESC) AS 앞사람,
    LEAD(SALARY) OVER (ORDER BY SALARY DESC) AS 뒷사람
FROM TEACHER
ORDER BY SALARY DESC;
